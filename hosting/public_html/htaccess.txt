# ==========================================================================
# سلسار — إعدادات Apache لجذر الويب
# ==========================================================================
# يُشحن باسم htaccess.txt لأن الأسماء التي تبدأ بنقطة تسقط في كثير من
# أدوات الضغط والرفع. بعد رفعه إلى الخادم أعد تسميته إلى:  .htaccess
#
# ملاحظتان على التصميم:
#   1) إعادة التوجيه إلى HTTPS مشروطة بـ X-Forwarded-Proto إلى جانب HTTPS،
#      لأن بعض الاستضافات المشتركة تُنهي SSL قبل Apache فلا تضبط HTTPS=on.
#      بدون هذا الشرط تنشأ حلقة إعادة توجيه لا نهائية.
#   2) حجب الملفات الداخلية لا يعتمد على mod_rewrite وحده، ويشمل .json
#      صراحةً — فملفات الكاش بهذا الامتداد وتحمل بيانات تشغيلية.
# ==========================================================================

Options -Indexes

# ---- إعادة التوجيه إلى HTTPS + حجب المسارات الداخلية ----
<IfModule mod_rewrite.c>
    RewriteEngine On

    # لا تُعد التوجيه إن كان الاتصال آمنًا أصلًا أو منتهيًا عند بروكسي أمامي
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !=https
    RewriteCond %{HTTP:X-Forwarded-SSL} !=on
    RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    RewriteRule ^private/ - [F,L]
    RewriteRule ^app-template\.html$ - [F,L]
</IfModule>

# ---- حجب بالاسم والامتداد: يعمل حتى لو كان mod_rewrite معطّلًا ----
<FilesMatch "^(\.env|\.git.*|config\.php|download|.*\.(json|log|bak|sql|ini|yml|yaml|md))$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

# ---- قالب التطبيق لا يُنزَّل مباشرة (index.php هو من يقدّمه بعد المصادقة) ----
<Files "app-template.html">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</Files>

# ---- رؤوس الأمان ----
<IfModule mod_headers.c>
    <FilesMatch "\.(php|html|js)$">
        Header always set Cache-Control "no-store, no-cache, must-revalidate, max-age=0"
        Header always set Pragma "no-cache"
        Header always set Expires "0"
    </FilesMatch>
    Header always set X-Content-Type-Options "nosniff"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Referrer-Policy "same-origin"
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
    Header always set Strict-Transport-Security "max-age=31536000"

    # ---- Content-Security-Policy (طبقة ثانية بعد تهريب المخرجات) ----
    # 'unsafe-inline' ضروري لأن التطبيق يعتمد على سكربتات وأنماط مضمّنة.
    # لذلك الحماية الحقيقية من XSS هي التهريب في app-template.html،
    # وهذا الرأس يمنع تحميل سكربتات خارجية وembeds وخطف base URL.
    #
    # إن تعطّل تسجيل الدخول أو الطباعة بعد تفعيله: احذف هذا السطر وحده
    # وأبلغني بما ظهر في Console — الباقي يبقى فعّالًا.
    Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://accounts.google.com https://cdn.sheetjs.com; style-src 'self' 'unsafe-inline' ; img-src 'self' data: https:; font-src 'self' data:; connect-src 'self' https://accounts.google.com https://oauth2.googleapis.com; frame-src https://accounts.google.com; frame-ancestors 'self'; object-src 'none'; base-uri 'none'; form-action 'self'"
</IfModule>
